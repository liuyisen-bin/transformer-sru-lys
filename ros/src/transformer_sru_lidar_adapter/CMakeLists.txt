cmake_minimum_required(VERSION 3.8)
project(transformer_sru_lidar_adapter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(SRU_STANDALONE "Build the projection library and geometry tests without ROS" OFF)
if(SRU_STANDALONE)
  find_package(Eigen3 REQUIRED)
  add_library(lidar_grid STATIC src/grid.cpp)
  set_target_properties(lidar_grid PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_include_directories(lidar_grid PUBLIC include)
  target_link_libraries(lidar_grid Eigen3::Eigen)
  add_library(lidar_projection_c SHARED src/grid_c_api.cpp)
  target_link_libraries(lidar_projection_c lidar_grid)
  enable_testing()
  add_executable(test_grid test/test_grid.cpp)
  target_link_libraries(test_grid lidar_grid)
  add_test(NAME lidar_grid_contract COMMAND test_grid)
  return()
endif()
find_package(ament_cmake REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(unitree_hg REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/LidarTensor.msg"
  DEPENDENCIES std_msgs builtin_interfaces)
add_library(lidar_grid src/grid.cpp)
set_target_properties(lidar_grid PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(lidar_grid PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(lidar_grid Eigen3::Eigen)
add_library(lidar_projection_c SHARED src/grid_c_api.cpp)
target_link_libraries(lidar_projection_c lidar_grid)
set_target_properties(lidar_projection_c PROPERTIES INSTALL_RPATH "$ORIGIN")
add_executable(lidar_adapter_node src/lidar_adapter_node.cpp)
target_link_libraries(lidar_adapter_node lidar_grid)
ament_target_dependencies(lidar_adapter_node rclcpp rcl_interfaces sensor_msgs std_msgs builtin_interfaces diagnostic_msgs unitree_hg)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(lidar_adapter_node "${cpp_typesupport_target}")
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_grid test/test_grid.cpp)
  target_link_libraries(test_grid lidar_grid)
  add_test(NAME lidar_grid_contract COMMAND test_grid)
endif()
install(TARGETS lidar_grid lidar_projection_c lidar_adapter_node
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
